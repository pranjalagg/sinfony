**Goal:** Build `sinfony`, a Python library that composes music from layered mathematics (curves + noise + music theory) and renders it to audio, MIDI, and annotated piano-roll pictures.

**Architecture:** Two core data types — continuous `Curve`s (intent: shape, tension, noise) and discrete `Events` (notes). A pipeline of small modules turns curves into notes: `Plan` (form, keys, tension) → harmony (`ChordTrack`) → voices (melody/bass/accompaniment/pad/loops, with rules and motifs) → `Perform` (velocity, timing, tempo curve) → renderers (numpy synth, MIDI, matplotlib). All randomness flows through a seeded `Rng` with independent child streams per layer.

**Tech Stack:** Python ≥ 3.11, uv, numpy, matplotlib, mido, pytest, ruff. Optional: sounddevice.

```python
"""Seeded randomness with independent per-layer streams."""

from __future__ import annotations

import hashlib
from collections.abc import Sequence
from typing import Any

import numpy as np


def _key(seed: int | str) -> int:
    digest = hashlib.sha256(str(seed).encode()).digest()
    return int.from_bytes(digest[:8], "little")


class Rng:
    """A seeded random stream.

    ``child(name)`` derives a new stream from this stream's seed and the name
    only. Drawing numbers in one layer therefore never changes another layer.
    """

    def __init__(self, seed: int | str = 0):
        self.seed = seed
        self.key = _key(seed)
        self.gen = np.random.default_rng(self.key)

    def child(self, name: str) -> Rng:
        return Rng(f"{self.key}/{name}")

    def random(self, size=None):
        return self.gen.random(size)

    def uniform(self, lo: float, hi: float, size=None):
        return self.gen.uniform(lo, hi, size)

    def integers(self, lo: int, hi: int, size=None):
        return self.gen.integers(lo, hi, size)

    def choice(self, items: Sequence[Any], p=None) -> Any:
        if len(items) == 0:
            raise ValueError("Cannot choose from an empty sequence")
        return items[int(self.gen.choice(len(items), p=p))]

    def __repr__(self) -> str:
        return f"Rng({self.seed!r})"
```

```python
"""Continuous curves over time (in beats) that compose like math."""

from __future__ import annotations

from collections.abc import Callable, Iterable

import numpy as np

ArrayFn = Callable[[np.ndarray], np.ndarray]


class Curve:
    """A vectorized function of time ``t`` (in beats).

    Curves combine with numbers and other curves, so
    ``0.6 * arch() + 0.3 * pink(rng)`` is itself a curve.
    """

    def __init__(self, f: ArrayFn, name: str = "curve"):
        self.f = f
        self.name = name

    def __call__(self, t):
        arr = np.asarray(t, dtype=float)
        out = np.broadcast_to(np.asarray(self.f(arr), dtype=float), arr.shape)
        if not np.all(np.isfinite(out)):
            raise ValueError(f"Curve '{self.name}' produced NaN or inf")
        if arr.ndim == 0:
            return float(out)
        return np.array(out)

    def __repr__(self) -> str:
        return f"Curve({self.name})"

    # --- arithmetic -----------------------------------------------------
    def _combine(self, other, op, sym: str, reverse: bool = False) -> Curve:
        o = lift(other)
        a, b = (o, self) if reverse else (self, o)
        return Curve(lambda t: op(a.f(t), b.f(t)), f"({a.name} {sym} {b.name})")

    def __add__(self, other):
        return self._combine(other, np.add, "+")

    def __radd__(self, other):
        return self._combine(other, np.add, "+", reverse=True)

    def __sub__(self, other):
        return self._combine(other, np.subtract, "-")

    def __rsub__(self, other):
        return self._combine(other, np.subtract, "-", reverse=True)

    def __mul__(self, other):
        return self._combine(other, np.multiply, "*")

    def __rmul__(self, other):
        return self._combine(other, np.multiply, "*", reverse=True)

    def __truediv__(self, other):
        return self._combine(other, np.divide, "/")

    def __rtruediv__(self, other):
        return self._combine(other, np.divide, "/", reverse=True)

    def __pow__(self, other):
        return self._combine(other, np.power, "**")

    def __rpow__(self, other):
        return self._combine(other, np.power, "**", reverse=True)

    def __neg__(self):
        return Curve(lambda t: -self.f(t), f"-{self.name}")

    def __abs__(self):
        return Curve(lambda t: np.abs(self.f(t)), f"|{self.name}|")

    # --- transforms -----------------------------------------------------
    def shift(self, dt: float) -> Curve:
        """Move the curve later in time by ``dt`` beats."""
        return Curve(lambda t: self.f(t - dt), f"{self.name}.shift({dt:g})")

    def stretch(self, k: float) -> Curve:
        """Slow the curve down by factor ``k`` (k > 1 makes it longer)."""
        if k <= 0:
            raise ValueError("stretch factor must be positive")
        return Curve(lambda t: self.f(t / k), f"{self.name}.stretch({k:g})")

    def clip(self, lo: float, hi: float) -> Curve:
        return Curve(lambda t: np.clip(self.f(t), lo, hi), f"{self.name}.clip")

    def remap(self, a: float, b: float, c: float, d: float) -> Curve:
        """Linear map of values from [a, b] to [c, d]."""
        if a == b:
            raise ValueError("remap needs a != b")
        return Curve(
            lambda t: c + (d - c) * (self.f(t) - a) / (b - a), f"{self.name}.remap"
        )

    def over(self, t0: float, t1: float) -> Curve:
        """Play a shape defined on [0, 1] across beats [t0, t1]; hold it outside."""
        if t1 <= t0:
            raise ValueError("over needs t1 > t0")
        return Curve(
            lambda t: self.f(np.clip((t - t0) / (t1 - t0), 0.0, 1.0)),
            f"{self.name}.over({t0:g},{t1:g})",
        )

    def tile(self, period: float) -> Curve:
        """Repeat a shape defined on [0, 1] every ``period`` beats."""
        if period <= 0:
            raise ValueError("tile period must be positive")
        return Curve(
            lambda t: self.f(np.mod(t, period) / period),
            f"{self.name}.tile({period:g})",
        )

    # --- inspection -----------------------------------------------------
    def sample(self, t0: float, t1: float, n: int = 256):
        t = np.linspace(t0, t1, n)
        return t, self(t)

    def show(self, t0: float = 0.0, t1: float = 1.0, n: int = 512):
        from .render.viz import plot_curve

        return plot_curve(self, t0, t1, n)


def lift(x) -> Curve:
    """Turn a number into a constant curve; pass curves through."""
    if isinstance(x, Curve):
        return x
    if isinstance(x, (int, float, np.integer, np.floating)):
        return const(float(x))
    raise TypeError(f"Cannot use {type(x).__name__} as a Curve")


# --- shapes -------------------------------------------------------------
def const(v: float) -> Curve:
    v = float(v)
    return Curve(lambda t: np.full(np.shape(t), v), f"{v:g}")


def sine(freq: float = 1.0, phase: float = 0.0) -> Curve:
    return Curve(lambda t: np.sin(2 * np.pi * (freq * t + phase)), f"sine({freq:g})")


def arch() -> Curve:
    """sin(pi*u) on [0, 1]: rises from 0 to 1 and falls back."""
    return Curve(lambda t: np.sin(np.pi * np.clip(t, 0.0, 1.0)), "arch")


def ramp() -> Curve:
    return Curve(lambda t: np.clip(t, 0.0, 1.0), "ramp")


def step(edges: Iterable[float], values: Iterable[float]) -> Curve:
    e = np.asarray(list(edges), dtype=float)
    v = np.asarray(list(values), dtype=float)
    if len(v) != len(e) + 1:
        raise ValueError("step needs len(values) == len(edges) + 1")
    if np.any(np.diff(e) <= 0):
        raise ValueError("step edges must be increasing")
    return Curve(lambda t: v[np.searchsorted(e, t, side="right")], "step")


def piecewise(points: Iterable[tuple[float, float]]) -> Curve:
    pts = sorted((float(x), float(y)) for x, y in points)
    if len(pts) < 2:
        raise ValueError("piecewise needs at least 2 points")
    xs = np.array([p[0] for p in pts])
    ys = np.array([p[1] for p in pts])
    if np.any(np.diff(xs) <= 0):
        raise ValueError("piecewise x values must be distinct")
    return Curve(lambda t: np.interp(t, xs, ys), "piecewise")
```

**Interfaces:**
- Consumes: `Curve` (Task 2), `Rng` (Task 1).
- Produces: `pink(rng, octaves=8, base=1/16) -> Curve`, `brown(rng, octaves=8, base=1/16) -> Curve`, `white(rng, rate=4.0) -> Curve`. All are deterministic for a given `Rng` state, have output clipped to `[-1, 1]` and std ≈ 0.5, and are named `"pink"`, `"brown"`, `"white"`. `base` is the lowest frequency in cycles per beat.

Background for the implementer: fractal value noise sums octaves at frequencies `base·2^o` with amplitude `gain^o`. Its spectrum is 1/f^β with β = 1 − 2·log₂(gain). Pink (β = 1) therefore needs `gain = 1` (equal amplitude per octave, Voss's construction). Brown (β = 2) needs `gain = 1/√2`.

```python
import numpy as np

from sinfony.curve import brown, pink, white
from sinfony.rng import Rng


def spectral_slope(curve, lo, hi, dt=1 / 32, n=2**17, segments=32):
    """Slope of log(power) vs log(frequency), averaged over segments."""
    x = curve(np.arange(n) * dt)
    x = x - x.mean()
    seg = x.reshape(segments, -1) * np.hanning(n // segments)
    power = (np.abs(np.fft.rfft(seg, axis=1)) ** 2).mean(axis=0)
    f = np.fft.rfftfreq(n // segments, dt)
    m = (f >= lo) & (f <= hi)
    return np.polyfit(np.log(f[m]), np.log(power[m]), 1)[0]


def test_pink_is_one_over_f():
    c = pink(Rng(0).child("t"), octaves=10, base=1 / 64)
    assert abs(spectral_slope(c, 1 / 16, 1.0) + 1.0) < 0.3


def test_brown_is_steeper_than_pink():
    c = brown(Rng(0).child("t"), octaves=10, base=1 / 64)
    assert abs(spectral_slope(c, 1 / 16, 1.0) + 2.0) < 0.4


def test_white_is_flat():
```

```python
# --- noise --------------------------------------------------------------
_U64 = np.uint64


def _hash01(i, salt: int) -> np.ndarray:
    """Deterministic pseudo-random value in [0, 1) per integer lattice point."""
    shape = np.shape(i)
    x = np.atleast_1d(np.asarray(i, dtype=np.int64)).view(np.uint64)
    with np.errstate(over="ignore"):
        x = x * _U64(0x9E3779B97F4A7C15)
        x ^= _U64(salt)
        x ^= x >> _U64(30)
        x *= _U64(0xBF58476D1CE4E5B9)
        x ^= x >> _U64(27)
        x *= _U64(0x94D049BB133111EB)
        x ^= x >> _U64(31)
    return ((x >> _U64(11)).astype(np.float64) / float(1 << 53)).reshape(shape)


def _value_noise(t, freq: float, salt: int) -> np.ndarray:
    """Random values on an integer lattice, smoothly (cosine) interpolated."""
    x = np.asarray(t, dtype=float) * freq
    i0 = np.floor(x)
    u = x - i0
    a = _hash01(i0, salt)
    b = _hash01(i0 + 1, salt)
    s = 0.5 - 0.5 * np.cos(np.pi * u)
    return (a + (b - a) * s) * 2.0 - 1.0


def _fractal(rng: Rng, octaves: int, base: float, gain: float, name: str) -> Curve:
    if octaves < 1:
        raise ValueError("octaves must be >= 1")
    if base <= 0:
        raise ValueError("base frequency must be positive")
    salts = [int(s) for s in rng.integers(0, 2**62, size=octaves)]
    amps = [gain**o for o in range(octaves)]
    # Each interpolated octave has std ~0.5; dividing by the root-sum-square of
    # the amplitudes keeps the total std ~0.5 as well.
    norm = float(np.sqrt(sum(a * a for a in amps)))

    def f(t):
        acc = 0.0
        for o, (salt, amp) in enumerate(zip(salts, amps, strict=True)):
            acc = acc + amp * _value_noise(t, base * 2.0**o, salt)
        return np.clip(acc / norm, -1.0, 1.0)

    return Curve(f, name)


def pink(rng: Rng, octaves: int = 8, base: float = 1 / 16) -> Curve:
    """1/f noise: equal amplitude in every octave (Voss's construction).

    Real melodies and loudness fluctuate with roughly this spectrum
    (Voss & Clarke, 1975): more structured than white noise, less
    sluggish than brown noise.
    """
    return _fractal(rng, octaves, base, 1.0, "pink")


def brown(rng: Rng, octaves: int = 8, base: float = 1 / 16) -> Curve:
    """1/f^2 noise: amplitude falls by 1/sqrt(2) per octave."""
    return _fractal(rng, octaves, base, 2**-0.5, "brown")


def white(rng: Rng, rate: float = 4.0) -> Curve:
    """A fresh independent value every 1/rate beats (sample-and-hold)."""
    if rate <= 0:
        raise ValueError("rate must be positive")
    salt = int(rng.integers(0, 2**62))
    return Curve(
        lambda t: _hash01(np.floor(np.asarray(t, dtype=float) * rate), salt) * 2.0
        - 1.0,
        "white",
    )
```

```python
"""Music theory as mathematics: pitch, scales, chords, voice leading, consonance."""

from __future__ import annotations

from collections.abc import Iterable
from dataclasses import dataclass

NOTE_NAMES = ("C", "C#", "D", "Eb", "E", "F", "F#", "G", "Ab", "A", "Bb", "B")


def freq(n: float) -> float:
    """Equal temperament: each semitone multiplies frequency by 2^(1/12)."""
    return 440.0 * 2.0 ** ((n - 69) / 12)


def pc(n: int) -> int:
    """Pitch class: the pitch modulo the octave (12 semitones)."""
    return int(n) % 12


def note_name(n: int) -> str:
    return f"{NOTE_NAMES[pc(n)]}{int(n) // 12 - 1}"


MODES: dict[str, tuple[int, ...]] = {
    "major": (0, 2, 4, 5, 7, 9, 11),
    "ionian": (0, 2, 4, 5, 7, 9, 11),
    "dorian": (0, 2, 3, 5, 7, 9, 10),
    "phrygian": (0, 1, 3, 5, 7, 8, 10),
    "lydian": (0, 2, 4, 6, 7, 9, 11),
    "mixolydian": (0, 2, 4, 5, 7, 9, 10),
    "aeolian": (0, 2, 3, 5, 7, 8, 10),
    "minor": (0, 2, 3, 5, 7, 8, 10),
    "locrian": (0, 1, 3, 5, 6, 8, 10),
    "pentatonic": (0, 2, 4, 7, 9),
}


@dataclass(frozen=True)
class Scale:
    """A tonic (MIDI pitch) plus the semitone steps above it.

    The major scale is ``Scale(60, (0, 2, 4, 5, 7, 9, 11))``: C major.
    """

    root: int
    steps: tuple[int, ...]

    def __post_init__(self):
        steps = tuple(sorted({int(s) % 12 for s in self.steps}))
        if not steps or steps[0] != 0:
            raise ValueError("Scale steps must include 0 (the tonic)")
        object.__setattr__(self, "steps", steps)
        object.__setattr__(self, "root", int(self.root))

    # --- constructors ---------------------------------------------------
    @classmethod
    def generated(cls, gen: int = 7, n: int = 7, root: int = 60) -> Scale:
        """Stack ``n`` copies of the interval ``gen`` (mod 12). gen=7 stacks fifths.

        For n >= 7 the stack starts one step below the tonic so the result is
        the major scale (F C G D A E B); for smaller n it starts on the tonic
        (C G D A E = major pentatonic).
        """
        if n < 1:
            raise ValueError("n must be >= 1")
        start = -1 if n >= 7 else 0
        return cls(root, tuple((gen * (k + start)) % 12 for k in range(n)))

    @classmethod
    def mode(cls, name: str, root: int = 60) -> Scale:
        if name not in MODES:
            raise ValueError(f"Unknown mode {name!r}; choose from {sorted(MODES)}")
        return cls(root, MODES[name])

    @classmethod
    def major(cls, root: int = 60) -> Scale:
        return cls.mode("major", root)

    @classmethod
    def minor(cls, root: int = 57) -> Scale:
        return cls.mode("minor", root)

    @classmethod
    def dorian(cls, root: int = 62) -> Scale:
        return cls.mode("dorian", root)

    @classmethod
    def phrygian(cls, root: int = 64) -> Scale:
        return cls.mode("phrygian", root)

    @classmethod
    def lydian(cls, root: int = 65) -> Scale:
        return cls.mode("lydian", root)

    @classmethod
    def mixolydian(cls, root: int = 67) -> Scale:
        return cls.mode("mixolydian", root)

    @classmethod
    def aeolian(cls, root: int = 57) -> Scale:
        return cls.mode("aeolian", root)

    @classmethod
    def locrian(cls, root: int = 59) -> Scale:
        return cls.mode("locrian", root)

    @classmethod
    def pentatonic(cls, root: int = 60) -> Scale:
        return cls.mode("pentatonic", root)

    # --- properties -----------------------------------------------------
    @property
    def pcs(self) -> frozenset[int]:
        return frozenset((self.root + s) % 12 for s in self.steps)

    @property
    def name(self) -> str:
        for name, steps in MODES.items():
            if steps == self.steps:
                return name
        return "custom"

    def is_minor(self) -> bool:
        return self.steps == MODES["aeolian"]

    def __len__(self) -> int:
        return len(self.steps)

    # --- pitch arithmetic -----------------------------------------------
    def degree(self, i: int) -> int:
        """MIDI pitch of scale degree ``i`` (0 = tonic, 7 = tonic an octave up)."""
        octave, idx = divmod(int(i), len(self.steps))
        return self.root + 12 * octave + self.steps[idx]

    def contains(self, pitch: int) -> bool:
        return (int(pitch) - self.root) % 12 in self.steps

    def nearest(self, pitch: float) -> int:
        p = int(round(pitch))
        cands = [q for q in range(p - 6, p + 7) if self.contains(q)]
        return min(cands, key=lambda q: (abs(q - pitch), q))

    def degree_of(self, pitch: int) -> int:
        q = self.nearest(pitch)
        octave, rel = divmod(q - self.root, 12)
        return octave * len(self.steps) + self.steps.index(rel)

    def pitches(self, lo: int, hi: int) -> list[int]:
        return [p for p in range(lo, hi + 1) if self.contains(p)]

    def transpose(self, k: int) -> Scale:
        return Scale(self.root + k, self.steps)

    def subset(self, degrees: Iterable[int]) -> Scale:
        return Scale(self.root, tuple(self.steps[d % len(self.steps)] for d in degrees))
```

```python
"""Discrete notes and the musical group actions on them."""

from __future__ import annotations

from collections.abc import Callable, Iterable, Iterator, Sequence
from dataclasses import dataclass, replace
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from .theory import Scale


@dataclass(frozen=True)
class Note:
    onset: float
    dur: float
    pitch: int
    vel: float = 0.7
    voice: str = "melody"

    @property
    def end(self) -> float:
        return self.onset + self.dur


def _sort_key(n: Note):
    return (n.onset, n.pitch, n.voice)


class Events(Sequence[Note]):
    """An immutable, time-sorted collection of notes with a time span."""

    def __init__(
        self, notes: Iterable[Note] = (), span: tuple[float, float] | None = None
    ):
        self._notes = tuple(sorted(notes, key=_sort_key))
        if span is None:
            if self._notes:
                span = (
                    min(n.onset for n in self._notes),
                    max(n.end for n in self._notes),
                )
            else:
                span = (0.0, 0.0)
        self.span = (float(span[0]), float(span[1]))

    # --- sequence protocol ----------------------------------------------
    def __len__(self) -> int:
        return len(self._notes)

    def __getitem__(self, i):
        return self._notes[i]

    def __iter__(self) -> Iterator[Note]:
        return iter(self._notes)

    def __eq__(self, other) -> bool:
        return (
            isinstance(other, Events)
            and self._notes == other._notes
            and self.span == other.span
        )

    __hash__ = None  # mutable-looking value semantics; not hashable

    def __repr__(self) -> str:
        return f"Events({len(self)} notes, span={self.span})"

    @property
    def length(self) -> float:
        return self.span[1] - self.span[0]

    @property
    def pitches(self) -> list[int]:
        return [n.pitch for n in self]

    # --- generic --------------------------------------------------------
    def map(self, fn: Callable[[Note], Note]) -> Events:
        return Events((fn(n) for n in self), self.span)

    def filter(self, pred: Callable[[Note], bool]) -> Events:
        return Events((n for n in self if pred(n)), self.span)

    # --- group actions --------------------------------------------------
    def transpose(self, k: int) -> Events:
        """Add ``k`` semitones to every pitch."""
        return self.map(lambda n: replace(n, pitch=n.pitch + int(k)))

    def transpose_diatonic(self, k: int, scale: Scale) -> Events:
        """Move every pitch ``k`` steps along ``scale`` (stays in key)."""
        return self.map(
            lambda n: replace(n, pitch=scale.degree(scale.degree_of(n.pitch) + k))
        )

    def invert(self, axis: float) -> Events:
        """Mirror pitches around ``axis``: p -> 2*axis - p."""
        twice = int(round(2 * axis))
        return self.map(lambda n: replace(n, pitch=twice - n.pitch))

    def retrograde(self) -> Events:
        """Play backwards in time within the span."""
        s, e = self.span
        return self.map(lambda n: replace(n, onset=s + e - n.end))

    def augment(self, k: float) -> Events:
        """Scale time by ``k`` around the span start (k=2 is twice as slow)."""
        if k <= 0:
            raise ValueError("augment factor must be positive")
        s, e = self.span
        notes = (replace(n, onset=s + (n.onset - s) * k, dur=n.dur * k) for n in self)
        return Events(notes, (s, s + (e - s) * k))

    def shift(self, dt: float) -> Events:
        s, e = self.span
        return Events((replace(n, onset=n.onset + dt) for n in self), (s + dt, e + dt))

    def __add__(self, other: Events) -> Events:
        """Concatenate: ``other`` starts where ``self`` ends."""
        moved = other.shift(self.span[1] - other.span[0])
        return Events(self._notes + moved._notes, (self.span[0], moved.span[1]))

    def __or__(self, other: Events) -> Events:
        """Overlay both sets of notes."""
        spans = [e.span for e in (self, other) if len(e) or e.span != (0.0, 0.0)]
        span = (
            (min(s for s, _ in spans), max(e for _, e in spans))
            if spans
            else (0.0, 0.0)
        )
        return Events(self._notes + other._notes, span)

    def by_voice(self) -> dict[str, Events]:
        groups: dict[str, list[Note]] = {}
        for n in self:
            groups.setdefault(n.voice, []).append(n)
        return {v: Events(ns, self.span) for v, ns in groups.items()}
```

```python
_ROMAN_RE = re.compile(
    r"^(?P<acc>[b#]?)"
    r"(?P<num>VII|VI|IV|V|III|II|I|vii|vi|iv|v|iii|ii|i)"
    r"(?P<qual>°|o|\+)?(?P<sev>7)?$"
)


def parse_roman(numeral: str, scale: Scale) -> Chord:
    """Chord for a Roman numeral in ``scale``. Case sets quality: V major, v minor."""
    text = numeral.strip()
    if "/" in text:
        head, _, tail = text.partition("/")
        target = parse_roman(tail, scale)
        chord = parse_roman(head, Scale.major(60 + target.root))
        return Chord(chord.root, chord.pcs, text)
    m = _ROMAN_RE.match(text)
    if not m:
        raise ValueError(f"Unknown Roman numeral: {numeral!r}")
    if len(scale) != 7:
        raise ValueError("Roman numerals need a 7-note scale")
    num = m["num"]
    root = pc(scale.degree(ROMAN.index(num.upper()))) + {"b": -1, "#": 1, "": 0}[
        m["acc"]
    ]
    seventh_ = m["sev"] is not None
    if m["qual"] in ("°", "o"):
        quality = "hdim7" if seventh_ else "dim"
    elif m["qual"] == "+":
        quality = "aug"
    elif num.isupper():
        quality = "dom7" if seventh_ else "maj"
    else:
        quality = "min7" if seventh_ else "min"
    return Chord.build(root, quality, text)


def numeral_function(numeral: str) -> str:
    """Harmonic function: T (tonic), S (pre-dominant), D (dominant), or other."""
    if "/" in numeral:
        return "S"  # secondary dominants prepare the dominant
    if numeral[:1] in ("b", "#"):
        return "other"
    head = numeral.rstrip("7°o+").upper()
    return {"I": "T", "VI": "T", "III": "T", "IV": "S", "II": "S", "V": "D", "VII": "D"}.get(
        head, "other"
    )

def _major_or_minor(chord: Chord) -> tuple[int, str]:
    if chord.quality not in ("maj", "min"):
        raise ValueError(f"P/L/R need a major or minor triad, got {chord}")
    return chord.root, chord.quality


def P(chord: Chord) -> Chord:
    """Parallel: C major <-> C minor (move the third by a semitone)."""
    r, q = _major_or_minor(chord)
    return Chord.build(r, "min" if q == "maj" else "maj")


def R(chord: Chord) -> Chord:
    """Relative: C major <-> A minor (move the fifth up a tone)."""
    r, q = _major_or_minor(chord)
    return Chord.build(r + 9, "min") if q == "maj" else Chord.build(r + 3, "maj")


def L(chord: Chord) -> Chord:
    """Leading-tone exchange: C major <-> E minor (move the root down a semitone)."""
    r, q = _major_or_minor(chord)
    return Chord.build(r + 4, "min") if q == "maj" else Chord.build(r + 8, "maj")
```

```python
# --- voice leading --------------------------------------------------------
def close_voicing(
    chord: Chord, n: int, lo: int, hi: int, center: float | None = None
) -> list[int]:
    """``n`` adjacent chord tones in [lo, hi] whose average sits nearest ``center``."""
    if lo >= hi:
        raise ValueError(f"register needs lo < hi, got {lo}-{hi}")
    pool = chord.pitches(lo, hi)
    need = min(n, len(chord.pcs))
    center = (lo + hi) / 2 if center is None else center
    best: tuple[float, list[int]] | None = None
    for i in range(len(pool) - n + 1):
        window = pool[i : i + n]
        if len({pc(p) for p in window}) < need:
            continue
        key = abs(sum(window) / n - center)
        if best is None or key < best[0]:
            best = (key, window)
    if best is None:
        raise ValueError(f"register {lo}-{hi} too narrow for {n} voices of {chord}")
    return list(best[1])


def voice_lead(voicing: list[int], target: Chord, lo: int, hi: int) -> list[int]:
    """Move each voice to a tone of ``target`` with the least total motion.

    Brute force over the 4 nearest target tones per voice. Ranking: cover as
    many of the chord's pitch classes as possible, then minimize total
    semitone motion, then prefer voices moving in opposite directions.
    """
    if lo >= hi:
        raise ValueError(f"register needs lo < hi, got {lo}-{hi}")
    pool = target.pitches(lo, hi)
    if not pool:
        raise ValueError(f"no tones of {target} in register {lo}-{hi}")
    voicing = sorted(voicing)
    options = [sorted(pool, key=lambda q, p=p: (abs(q - p), q))[:4] for p in voicing]
    need = min(len(voicing), len(target.pcs))
    best = None
    for combo in itertools.product(*options):
        moves = [q - p for p, q in zip(voicing, combo, strict=True)]
        covered = len({pc(q) for q in combo})
        cost = sum(abs(m) for m in moves)
        contrary = sum(1 for a, b in itertools.combinations(moves, 2) if a * b < 0)
        key = (covered < need, cost, -contrary, tuple(sorted(combo)))
        if best is None or key < best[0]:
            best = (key, combo)
    return sorted(best[1])


# --- consonance -----------------------------------------------------------
def _plomp_levelt(f1: float, f2: float, a1: float, a2: float) -> float:
    """Dissonance of two sine partials (Plomp-Levelt curve, Sethares' fit)."""
    b1, b2, d_star, s1, s2 = 3.5, 5.75, 0.24, 0.0207, 18.96
    s = d_star / (s1 * min(f1, f2) + s2)
    d = abs(f2 - f1)
    return min(a1, a2) * (np.exp(-b1 * s * d) - np.exp(-b2 * s * d))


def roughness(pitches: Iterable[int], partials: int = 6) -> float:
    """Sensory dissonance of notes with harmonic overtones (Sethares 1993).

    Each note contributes partials k*f with amplitude 0.88^(k-1); every pair
    of partials adds its Plomp-Levelt dissonance. Lower means smoother.
    """
    comps = [
        (freq(p) * k, 0.88 ** (k - 1)) for p in pitches for k in range(1, partials + 1)
    ]
    return float(
        sum(
            _plomp_levelt(f1, f2, a1, a2)
            for (f1, a1), (f2, a2) in itertools.combinations(comps, 2)
        )
    )
```

```python
from dataclasses import dataclass

import numpy as np

from .rng import Rng

EPS = 1e-9


@dataclass(frozen=True)
class Section:
    index: int
    label: str
    base_label: str
    start: float
    end: float
    bars: int

    @property
    def length(self) -> float:
        return self.end - self.start


class Form:
    """Sections laid end to end, e.g. ``Form("A A' B A", bars=8)``."""

    def __init__(
        self, spec: str, bars: int | list[int] = 8, meter: tuple[int, int] = (4, 4)
    ):
        labels = spec.split()
        if not labels:
            raise ValueError("Form spec is empty")
        num, den = meter
        if num < 1 or den not in (1, 2, 4, 8, 16):
            raise ValueError(f"Unsupported meter {meter}")
        bar_list = [bars] * len(labels) if isinstance(bars, int) else list(bars)
        if len(bar_list) != len(labels):
            raise ValueError("bars list must match the number of sections")
        if any(b < 1 for b in bar_list):
            raise ValueError("each section needs at least 1 bar")
        self.spec = spec
        self.meter = (num, den)
        self.beats_per_bar = num * 4 / den
        sections, t = [], 0.0
        for i, (label, b) in enumerate(zip(labels, bar_list, strict=True)):
            end = t + b * self.beats_per_bar
            sections.append(Section(i, label, label.rstrip("'"), t, end, b))
            t = end
        self.sections = tuple(sections)
        self.total_beats = t

    def section_at(self, t: float) -> Section:
        for s in self.sections:
            if t < s.end - EPS:
                return s
        return self.sections[-1]

    def __repr__(self) -> str:
        return f"Form({self.spec!r}, meter={self.meter}, beats={self.total_beats:g})"


def default_tension(form: Form, rng: Rng) -> Curve:
    """Rise to a peak around 65% of the piece, then resolve; plus faint pink noise."""
    T = form.total_beats
    shape = piecewise([(0, 0.2), (0.65 * T, 1.0), (0.9 * T, 0.45), (T, 0.05)])
    return (shape + 0.05 * pink(rng, base=1 / 8)).clip(0.0, 1.0)


def dominant_key(key: Scale) -> Scale:
    """The key a fifth up, written a fourth down so the register stays put."""
    return Scale(key.root - 5, key.steps)


def relative_key(key: Scale) -> Scale:
    """Major <-> minor sharing the same notes (C major <-> A minor)."""
    if key.is_minor():
        return Scale.major(key.root + 3)
    return Scale.minor(key.root - 3)


class Plan:
    """Everything decided before any note: seed, key areas, form, tempo, tension."""

    def __init__(
        self,
        seed: int | str = 0,
        key: Scale | None = None,
        form: Form | None = None,
        tempo: float | Curve = 90.0,
        tension: Curve | None = None,
        modulate: bool = False,
    ):
        if not isinstance(tempo, Curve) and tempo <= 0:
            raise ValueError("tempo must be positive")
        self.seed = seed
        self.rng = Rng(seed)
        self.key = key or Scale.major(60)
        self.form = form or Form("A A' B A")
        self.tempo = tempo if isinstance(tempo, Curve) else const(tempo)
        self.tension = (
            tension
            if tension is not None
            else default_tension(self.form, self.rng.child("tension"))
        )
        self._keys = self._plan_keys(modulate, self.rng.child("keys"))

    def _plan_keys(self, modulate: bool, rng: Rng) -> dict[str, Scale]:
        keys: dict[str, Scale] = {}
        for s in self.form.sections:
            if s.base_label in keys:
                continue
            key = self.key
            if modulate and s.base_label == "B":
                mean_tension = float(np.mean(self.tension(np.linspace(s.start, s.end, 16))))
                if rng.random() < mean_tension:
                    key = rng.choice([dominant_key(self.key), relative_key(self.key)])
            keys[s.base_label] = key
        return keys

    @property
    def total_beats(self) -> float:
        return self.form.total_beats

    def section_key(self, section) -> Scale:
        return self._keys[section.base_label]

    def key_at(self, t: float) -> Scale:
        return self.section_key(self.form.section_at(t))
```

```python
"""Harmony: chord progressions generated from functions, lattices, or by hand."""

from __future__ import annotations

from collections.abc import Callable, Iterable, Iterator, Sequence
from dataclasses import dataclass

import numpy as np

from .plan import Plan, Section
from .rng import Rng
from .theory import Chord, Scale, numeral_function, parse_roman


@dataclass(frozen=True)
class ChordSpan:
    onset: float
    dur: float
    chord: Chord
    function: str = "other"

    @property
    def end(self) -> float:
        return self.onset + self.dur


class ChordTrack(Sequence[ChordSpan]):
    """Chords laid out in time."""

    def __init__(self, spans: Iterable[ChordSpan]):
        self._spans = tuple(sorted(spans, key=lambda s: s.onset))
        if not self._spans:
            raise ValueError("ChordTrack needs at least one span")
        self._onsets = np.array([s.onset for s in self._spans])

    def __len__(self) -> int:
        return len(self._spans)

    def __getitem__(self, i):
        return self._spans[i]

    def __iter__(self) -> Iterator[ChordSpan]:
        return iter(self._spans)

    def __eq__(self, other) -> bool:
        return isinstance(other, ChordTrack) and self._spans == other._spans

    __hash__ = None

    def __repr__(self) -> str:
        return f"ChordTrack({len(self)} chords)"

    def at(self, t: float) -> ChordSpan:
        i = int(np.searchsorted(self._onsets, t + 1e-9, side="right")) - 1
        return self._spans[max(i, 0)]

    def show(self):
        from .render.viz import plot_chords

        return plot_chords(self)


MakeSection = Callable[[Section, Rng], list[ChordSpan]]


def by_section(plan: Plan, rng: Rng, make: MakeSection) -> ChordTrack:
    """Generate chords once per base label and reuse them for repeats."""
    cache: dict[str, list[ChordSpan]] = {}
    spans = []
    for sec in plan.form.sections:
        if sec.base_label not in cache:
            cache[sec.base_label] = make(sec, rng.child(f"section:{sec.base_label}"))
        for s in cache[sec.base_label]:
            spans.append(ChordSpan(s.onset + sec.start, s.dur, s.chord, s.function))
    return ChordTrack(spans)


class Fixed:
    """A hand-written progression of Roman numerals, repeated across the form."""

    def __init__(self, numerals: list[str], per_bar: int = 1):
        if not numerals:
            raise ValueError("Fixed needs at least one numeral")
        if per_bar < 1:
            raise ValueError("per_bar must be >= 1")
        for n in numerals:
            parse_roman(n, Scale.major(60))  # fail early on typos
        self.numerals = list(numerals)
        self.per_bar = per_bar

    def generate(self, plan: Plan, rng: Rng) -> ChordTrack:
        dur = plan.form.beats_per_bar / self.per_bar

        def make(sec: Section, _rng: Rng) -> list[ChordSpan]:
            key = plan.section_key(sec)
            out = []
            for i in range(sec.bars * self.per_bar):
                num = self.numerals[i % len(self.numerals)]
                out.append(
                    ChordSpan(i * dur, dur, parse_roman(num, key), numeral_function(num))
                )
            return out

        return by_section(plan, rng, make)
```

```python
# --- functional harmony -----------------------------------------------------
FUNCTIONS = {
    "major": {
        "T": [("I", 0.7), ("vi", 0.3)],
        "S": [("IV", 0.6), ("ii", 0.4)],
        "D": [("V", 0.8), ("vii°", 0.2)],
    },
    "minor": {
        "T": [("i", 0.7), ("VI", 0.3)],
        "S": [("iv", 0.6), ("ii°", 0.4)],
        "D": [("V", 0.8), ("vii°", 0.2)],
    },
}
# Probability of the next harmonic function given the current one.
TRANSITIONS = {
    "T": {"T": 0.2, "S": 0.5, "D": 0.3},
    "S": {"T": 0.1, "S": 0.2, "D": 0.7},
    "D": {"T": 0.8, "S": 0.05, "D": 0.15},
}

Slot = tuple[float, float, tuple[str, str] | None]


class Functional:
    """Tonic -> pre-dominant -> dominant -> tonic as a Markov chain, with cadences.

    Higher tension makes dominant chords (and secondary dominants V/V) more
    likely. Each section opens on the tonic, has a half cadence (V) in the
    middle, and closes with an authentic cadence (V -> I).
    """

    def __init__(self, rhythm: Curve | None = None):
        self.rhythm = rhythm

    def generate(self, plan: Plan, rng: Rng) -> ChordTrack:
        bpb = plan.form.beats_per_bar

        def make(sec: Section, srng: Rng) -> list[ChordSpan]:
            key = plan.section_key(sec)
            table = FUNCTIONS["minor" if key.is_minor() else "major"]
            tonic, pre, dom = table["T"][0][0], table["S"][0][0], "V"
            out: list[ChordSpan] = []
            prev_fn, prev_num = "T", None
            for onset, dur, forced in self._slots(sec, bpb, tonic, pre, dom):
                tension = float(plan.tension(sec.start + onset))
                if forced is not None:
                    num, fn = forced
                elif prev_num == "V/V":
                    num, fn = "V", "D"
                else:
                    fn = self._next_function(prev_fn, tension, srng)
                    num = self._pick(table[fn], prev_num, srng)
                    if fn == "S" and tension > 0.5 and srng.random() < tension - 0.5:
                        num = "V/V"
                out.append(ChordSpan(onset, dur, parse_roman(num, key), fn))
                prev_fn, prev_num = fn, num
            return out

        return self._prime(by_section(plan, rng, make), plan)

    def _slots(self, sec: Section, bpb: float, tonic: str, pre: str, dom: str):
        n = sec.bars
        cad_end = n - 1
        cad_half = n // 2 - 1 if n >= 4 else None
        T, S, D = (tonic, "T"), (pre, "S"), (dom, "D")
        slots: list[Slot] = []
        for b in range(n):
            t0 = b * bpb
            if b == cad_end:
                halves = [T]
            elif b == cad_half:
                halves = [D]
            elif b + 1 == cad_end:
                halves = [S, D]
            elif cad_half is not None and b + 1 == cad_half:
                halves = [None, S]
            elif self.rhythm is not None and float(self.rhythm(sec.start + t0)) >= 1.5:
                halves = [None, None]
            else:
                halves = [None]
            if b == 0 and b != cad_end:
                halves[0] = T
            dur = bpb / len(halves)
            slots += [(t0 + i * dur, dur, h) for i, h in enumerate(halves)]
        return slots

    @staticmethod
    def _next_function(prev: str, tension: float, rng: Rng) -> str:
        w = dict(TRANSITIONS[prev])
        w["D"] *= 1.0 + tension
        w["T"] *= 1.5 - tension
        names = list(w)
        p = np.array([w[k] for k in names])
        return rng.choice(names, p=p / p.sum())

    @staticmethod
    def _pick(options, prev: str | None, rng: Rng) -> str:
        opts = [(n, w) for n, w in options if n != prev] or list(options)
        names = [n for n, _ in opts]
        p = np.array([w for _, w in opts])
        return rng.choice(names, p=p / p.sum())

    @staticmethod
    def _prime(track: ChordTrack, plan: Plan) -> ChordTrack:
        """Primed sections (A') that are not last end deceptively on vi."""
        spans = list(track)
        last = len(plan.form.sections) - 1
        for sec in plan.form.sections:
            if sec.label == sec.base_label or sec.index == last:
                continue
            key = plan.section_key(sec)
            deceptive = "VI" if key.is_minor() else "vi"
            for i, s in enumerate(spans):
                if abs(s.end - sec.end) < 1e-9:
                    spans[i] = ChordSpan(s.onset, s.dur, parse_roman(deceptive, key), "T")
        return ChordTrack(spans)
```

```python
# --- Tonnetz walk -------------------------------------------------------------
def home_chord(key: Scale) -> Chord:
    """Tonic triad: major if the key has a major third above the tonic, else minor."""
    quality = "maj" if key.contains(key.root + 4) else "min"
    return Chord.build(key.root, quality)


class TonnetzWalk:
    """A random walk on the Tonnetz: each move is P, L, or R.

    These moves keep two common tones and shift one note by a step, so the
    harmony drifts smoothly. Moves that stay inside the key are preferred.
    """

    def __init__(self, rate: Curve | None = None, bars_per_chord: tuple[int, int] = (2, 4)):
        lo, hi = bars_per_chord
        if lo < 1 or hi < lo:
            raise ValueError("bars_per_chord needs 1 <= lo <= hi")
        self.rate = rate
        self.bars_per_chord = (lo, hi)

    def generate(self, plan: Plan, rng: Rng) -> ChordTrack:
        bpb = plan.form.beats_per_bar
        lo, hi = self.bars_per_chord

        def make(sec: Section, srng: Rng) -> list[ChordSpan]:
            key = plan.section_key(sec)
            chord = home_chord(key)
            t, out = 0.0, []
            while t < sec.length - 1e-9:
                bars = int(srng.integers(lo, hi + 1))
                dur = min(bars * bpb, sec.length - t)
                if out:
                    at = sec.start + t
                    p = (
                        float(self.rate(at))
                        if self.rate is not None
                        else 0.3 + 0.6 * float(plan.tension(at))
                    )
                    if srng.random() < p:
                        chord = self._move(chord, key, srng)
                out.append(ChordSpan(t, dur, chord, "other"))
                t += dur
            return out

        spans = list(by_section(plan, rng, make))
        last = spans[-1]
        home = home_chord(plan.section_key(plan.form.sections[-1]))
        spans[-1] = ChordSpan(last.onset, last.dur, home, last.function)
        return ChordTrack(spans)

    @staticmethod
    def _move(chord: Chord, key: Scale, rng: Rng) -> Chord:
        moves = [P(chord), L(chord), R(chord)]
        fits = [len(c.pcs & key.pcs) for c in moves]
        best = max(fits)
        return rng.choice([c for c, f in zip(moves, fits, strict=True) if f == best])
```

```python
"""Meter and rhythm generators."""

from __future__ import annotations

from collections.abc import Callable

import numpy as np

from .curve import Curve, lift
from .rng import Rng

EPS = 1e-6


class MeterGrid:
    """Metric hierarchy: downbeat > mid-bar > beat > subdivision > other."""

    def __init__(self, meter: tuple[int, int] = (4, 4)):
        num, den = meter
        self.meter = (num, den)
        self.beats_per_bar = num * 4 / den
        compound = den == 8 and num % 3 == 0 and num > 3
        self.beat = 1.5 if compound else 4 / den
        self.sub = self.beat / 3 if compound else self.beat / 2
        self._even = (num // 3 if compound else num) % 2 == 0 and num > 2

    @staticmethod
    def _on(x: float, unit: float) -> bool:
        r = x % unit
        return r < EPS or unit - r < EPS

    def strength(self, t: float) -> float:
        pos = float(t) % self.beats_per_bar
        if self._on(pos, self.beats_per_bar):
            return 1.0
        if self._even and self._on(pos, self.beats_per_bar / 2):
            return 0.75
        if self._on(pos, self.beat):
            return 0.5
        if self._on(pos, self.sub):
            return 0.25
        return 0.1


Rhythm = Callable[[float, float, MeterGrid, Rng], list[tuple[float, float]]]


def _fill(onsets: list[float], end: float, cap: float | None) -> list[tuple[float, float]]:
    """Each note lasts until the next onset (or ``end``), optionally capped."""
    out = []
    for j, t in enumerate(onsets):
        nxt = onsets[j + 1] if j + 1 < len(onsets) else end
        d = nxt - t
        out.append((t, min(d, cap) if cap is not None else d))
    return out


class _Every:
    def __init__(self, step: float, dur: float | None):
        if step <= 0:
            raise ValueError("step must be positive")
        self.step, self.dur = step, dur

    def __call__(self, start, end, grid, rng):
        n = int(np.ceil((end - start) / self.step - EPS))
        return [(start + i * self.step, self.dur or self.step) for i in range(n)]


def every(step: float = 1.0, dur: float | None = None) -> Rhythm:
    """A note every ``step`` beats."""
    return _Every(step, dur)


def euclid_pattern(k: int, n: int, rotate: int = 0) -> list[int]:
    """``k`` hits spread as evenly as possible over ``n`` steps.

    Uses the Bresenham form ``(i*k) mod n < k``, which yields Bjorklund's
    Euclidean rhythms up to rotation. E(3, 8) is the Cuban tresillo.
    """
    if n < 1 or not 0 <= k <= n:
        raise ValueError("euclid needs n >= 1 and 0 <= k <= n")
    pat = [1 if (i * k) % n < k else 0 for i in range(n)]
    r = rotate % n
    return pat[r:] + pat[:r]


class _Euclid:
    def __init__(self, k, n, rotate, step, dur):
        if step <= 0:
            raise ValueError("step must be positive")
        self.pattern = euclid_pattern(k, n, rotate)
        self.step, self.dur = step, dur

    def __call__(self, start, end, grid, rng):
        n = len(self.pattern)
        first = int(np.ceil(start / self.step - EPS))
        last = int(np.ceil(end / self.step - EPS))
        onsets = [i * self.step for i in range(first, last) if self.pattern[i % n]]
        if self.dur is None:
            return _fill(onsets, end, None)
        return [(t, min(self.dur, end - t)) for t in onsets]


def euclid(
    k: int, n: int, rotate: int = 0, step: float = 0.5, dur: float | None = None
) -> Rhythm:
    """Euclidean rhythm E(k, n) on a grid of ``step`` beats, phased from beat 0."""
    return _Euclid(k, n, rotate, step, dur)


class _Density:
    def __init__(self, curve: Curve, step: float, max_dur: float | None):
        if step <= 0:
            raise ValueError("step must be positive")
        self.curve, self.step, self.max_dur = curve, step, max_dur

    def __call__(self, start, end, grid, rng):
        n = int(np.ceil((end - start) / self.step - EPS))
        onsets = []
        for i in range(n):
            t = start + i * self.step
            p = float(np.clip(self.curve(t), 0.0, 1.0)) * grid.strength(t)
            if rng.random() < p:  # always draw, so the stream stays aligned
                onsets.append(t)
        return _fill(onsets, end, self.max_dur)


def density(curve, step: float = 0.5, max_dur: float | None = None) -> Rhythm:
    """Onset probability at each grid step = density(t) x beat strength(t)."""
    return _Density(lift(curve), step, max_dur)
```

```python
"""Melody rules: small penalty functions that shape note choices."""

from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass

from .theory import Chord, Scale, pc


@dataclass(frozen=True)
class Context:
    """What a rule can see when judging a candidate pitch."""

    prev: tuple[int, ...]
    chord: Chord
    scale: Scale
    strength: float
    t: float
    section_end: bool
    cadence: str  # "authentic" or "none"


Rule = Callable[[Context, int], float]


def _named(rule: Rule, name: str) -> Rule:
    rule.__name__ = name
    return rule


def step_bias(w: float = 1.5) -> Rule:
    """Penalize leaps: w per semitone beyond a whole step."""

    def rule(ctx: Context, c: int) -> float:
        return w * max(0, abs(c - ctx.prev[-1]) - 2) if ctx.prev else 0.0

    return _named(rule, f"step_bias({w:g})")


def gap_fill(w: float = 4.0) -> Rule:
    """After a leap (> 4 semitones), penalize moving further the same way."""

    def rule(ctx: Context, c: int) -> float:
        if len(ctx.prev) < 2:
            return 0.0
        leap = ctx.prev[-1] - ctx.prev[-2]
        move = c - ctx.prev[-1]
        return w if abs(leap) > 4 and move * leap > 0 else 0.0

    return _named(rule, f"gap_fill({w:g})")


def range_limit(lo: int, hi: int, w: float = 100.0) -> Rule:
    def rule(ctx: Context, c: int) -> float:
        return w if c < lo or c > hi else 0.0

    return _named(rule, f"range_limit({lo},{hi})")


def avoid_repeat(w: float = 2.0, n: int = 2) -> Rule:
    """Penalize playing the same pitch more than ``n`` times in a row."""

    def rule(ctx: Context, c: int) -> float:
        recent = ctx.prev[-n:]
        return w if len(recent) == n and all(p == c for p in recent) else 0.0

    return _named(rule, f"avoid_repeat({w:g},{n})")


def resolve_leading_tone(w: float = 3.0) -> Rule:
    """The note a semitone below the tonic should rise to the tonic."""

    def rule(ctx: Context, c: int) -> float:
        if not ctx.prev:
            return 0.0
        last = ctx.prev[-1]
        if pc(last) == pc(ctx.scale.root - 1) and c != last + 1:
            return w
        return 0.0

    return _named(rule, f"resolve_leading_tone({w:g})")


def end_on_tonic(w: float = 20.0) -> Rule:
    """At an authentic cadence, the last note of the section should be the tonic."""

    def rule(ctx: Context, c: int) -> float:
        if ctx.section_end and ctx.cadence == "authentic" and pc(c) != pc(ctx.scale.root):
            return w
        return 0.0

    return _named(rule, f"end_on_tonic({w:g})")
```

```python
"""Voices: turn chords and curves into notes."""

from __future__ import annotations

import warnings
from collections.abc import Callable
from dataclasses import dataclass, field, replace

import numpy as np

from .curve import Curve
from .events import Events, Note
from .harmony import ChordTrack
from .plan import Plan, Section
from .rhythm import MeterGrid, Rhythm
from .rng import Rng
from .rules import Context, Rule
from .theory import Chord, Scale, close_voicing, pc, voice_lead

EPS = 1e-6
STRONG = 0.5
ROLES = ("melody", "bass", "accomp", "pad", "loop")
PATTERNS = ("block", "sustain", "arpeggio", "alberti")
ORDER = {"arpeggio": (0, 1, 2, 1), "alberti": (0, 2, 1, 2)}


@dataclass
class Voice:
    """One line of music: what it plays (role), where (register), and how."""

    name: str
    role: str
    register: tuple[int, int]
    rhythm: Rhythm | None = None
    contour: Curve | None = None
    rules: list[Rule] = field(default_factory=list)
    pattern: str | dict[str, str] = "block"
    step: float = 0.5
    temperature: float = 0.0
    motif: bool = False
    subset: tuple[int, ...] | None = None
    cell: tuple[int, ...] = (0, 2, 4)
    vel: float = 0.7

    def __post_init__(self):
        lo, hi = self.register
        if lo >= hi:
            raise ValueError(
                f"Voice {self.name!r}: register needs lo < hi, got {self.register}"
            )
        if self.role not in ROLES:
            raise ValueError(f"Voice {self.name!r}: unknown role {self.role!r}; use {ROLES}")
        if self.role == "melody" and (self.rhythm is None or self.contour is None):
            raise ValueError(f"Voice {self.name!r}: melody voices need rhythm and contour")
        if self.role == "loop" and self.rhythm is None:
            raise ValueError(f"Voice {self.name!r}: loop voices need a rhythm")
        if not self.cell:
            raise ValueError(f"Voice {self.name!r}: cell must not be empty")


# --- helpers ------------------------------------------------------------------
def fold(p: int, lo: int, hi: int) -> int:
    """Move ``p`` by octaves into [lo, hi] (clamp if the register is < an octave)."""
    if hi - lo < 12:
        return min(max(p, lo), hi)
    while p < lo:
        p += 12
    while p > hi:
        p -= 12
    return p


def cadence_of(chord: Chord, key: Scale, is_end: bool) -> str:
    if is_end and chord.root == pc(key.root) and chord.quality in ("maj", "min"):
        return "authentic"
    return "none"


def candidates(
    chord: Chord,
    key: Scale,
    strength: float,
    lo: int,
    hi: int,
    subset: tuple[int, ...] | None,
    name: str,
) -> list[int]:
    """Pitches a voice may play here: chord tones on strong beats, else scale tones too."""
    for widen in (0, 12):
        a, b = lo - widen, hi + widen
        tones = chord.pitches(a, b)
        pool = tones if strength >= STRONG else sorted(set(key.pitches(a, b)) | set(tones))
        if subset:
            allowed = key.subset(subset).pcs
            filtered = [p for p in pool if pc(p) in allowed]
            pool = filtered or pool
        if pool:
            if widen:
                warnings.warn(
                    f"Voice {name!r}: no candidate in register {lo}-{hi}; widened by an octave",
                    UserWarning,
                    stacklevel=2,
                )
            return pool
    raise ValueError(f"Voice {name!r}: no candidate pitches near register {lo}-{hi}")


def choose(cands: list[int], scores: np.ndarray, temperature: float, rng: Rng) -> int:
    """Lowest score wins; with temperature > 0, sample from softmax(-score/T)."""
    if temperature <= 0:
        return cands[int(np.argmin(scores))]
    z = np.exp(-(scores - scores.min()) / temperature)
    return int(rng.choice(cands, p=z / z.sum()))


def requantize(
    ev: Events,
    plan: Plan,
    chords: ChordTrack,
    grid: MeterGrid,
    register: tuple[int, int],
    subset: tuple[int, ...] | None = None,
    name: str = "voice",
) -> Events:
    """Fit notes to the actual harmony: nearest allowed pitch, tonic at cadences."""
    lo, hi = register
    out = []
    for n in ev:
        sec = plan.form.section_at(n.onset)
        key = plan.section_key(sec)
        chord = chords.at(n.onset).chord
        p = fold(n.pitch, lo, hi)
        cands = candidates(chord, key, grid.strength(n.onset), lo, hi, subset, name)
        if n.end >= sec.end - EPS and cadence_of(chord, key, True) == "authentic":
            tonics = [c for c in chord.pitches(lo, hi) if pc(c) == pc(key.root)]
            cands = tonics or cands
        out.append(replace(n, pitch=min(cands, key=lambda c: (abs(c - p), c))))
    return Events(out, ev.span)


# --- melody ---------------------------------------------------------------------
def realize_melody(
    voice: Voice,
    plan: Plan,
    chords: ChordTrack,
    grid: MeterGrid,
    rng: Rng,
    sec: Section,
    start: float | None = None,
    end: float | None = None,
) -> Events:
    """Contour gives a target pitch; candidates are scored by distance + rule penalties."""
    start = sec.start if start is None else start
    end = sec.end if end is None else end
    lo, hi = voice.register
    key = plan.section_key(sec)
    notes: list[Note] = []
    prev: list[int] = []
    for onset, dur in voice.rhythm(start, end, grid, rng):
        span = chords.at(onset)
        strength = grid.strength(onset)
        cands = candidates(span.chord, key, strength, lo, hi, voice.subset, voice.name)
        is_end = onset + dur >= sec.end - EPS
        ctx = Context(
            tuple(prev), span.chord, key, strength, onset, is_end,
            cadence_of(span.chord, key, is_end),
        )
        target = lo + float(voice.contour(onset)) * (hi - lo)
        scores = np.array(
            [abs(c - target) + sum(r(ctx, c) for r in voice.rules) for c in cands]
        )
        pitch = choose(cands, scores, voice.temperature, rng)
        notes.append(Note(onset, dur, pitch, voice.vel, voice.name))
        prev.append(pitch)
    return Events(notes, (start, end))


MakeNotes = Callable[[Voice, Plan, ChordTrack, MeterGrid, Rng, Section], Events]


def per_section(
    voice: Voice,
    plan: Plan,
    chords: ChordTrack,
    grid: MeterGrid,
    rng: Rng,
    make: MakeNotes,
) -> Events:
    """Generate once per base label, copy into repeats, then fit to the chords."""
    cache: dict[str, Events] = {}
    notes: list[Note] = []
    for sec in plan.form.sections:
        if sec.base_label not in cache:
            fresh = make(voice, plan, chords, grid, rng.child(f"section:{sec.base_label}"), sec)
            cache[sec.base_label] = fresh.shift(-sec.start)
        notes.extend(cache[sec.base_label].shift(sec.start))
    ev = Events(notes, (0.0, plan.total_beats))
    return requantize(ev, plan, chords, grid, voice.register, voice.subset, voice.name)


# --- bass, chords, loops ----------------------------------------------------------
def realize_bass(voice, plan, chords, grid, rng) -> Events:
    """Chord roots, with inversions when they shorten the bass line (never at cadences)."""
    lo, hi = voice.register
    ends = {round(s.end, 6) for s in plan.form.sections}
    notes, prev = [], None
    for span in chords:
        tones = span.chord.pitches(lo, hi) or span.chord.pitches(lo - 12, hi + 12)
        ref = prev if prev is not None else (lo + hi) / 2 - 4
        cadence = round(span.end, 6) in ends
        options = [
            (abs(p - ref) + (0 if pc(p) == span.chord.root else 3), p)
            for p in tones
            if pc(p) == span.chord.root or not cadence
        ] or [(abs(p - ref), p) for p in tones]
        pitch = min(options)[1]
        hits = voice.rhythm(span.onset, span.end, grid, rng) if voice.rhythm else [
            (span.onset, span.dur)
        ]
        notes += [Note(o, d, pitch, voice.vel, voice.name) for o, d in hits]
        prev = pitch
    return Events(notes, (0.0, plan.total_beats))


def _pattern_for(voice: Voice, base_label: str) -> str:
    if voice.role == "pad":
        return "sustain"
    kind = (
        voice.pattern.get(base_label, "block")
        if isinstance(voice.pattern, dict)
        else voice.pattern
    )
    if kind not in PATTERNS:
        raise ValueError(f"Voice {voice.name!r}: unknown pattern {kind!r}; use {PATTERNS}")
    return kind


def realize_chords(voice, plan, chords, grid, rng) -> Events:
    """Voice-led chords laid out as sustain, block, arpeggio, or Alberti bass."""
    lo, hi = voice.register
    n_voices = 4 if voice.role == "pad" else 3
    voicing, notes = None, []
    for span in chords:
        voicing = (
            close_voicing(span.chord, n_voices, lo, hi)
            if voicing is None
            else voice_lead(voicing, span.chord, lo, hi)
        )
        kind = _pattern_for(voice, plan.form.section_at(span.onset).base_label)
        if kind in ("sustain", "block"):
            strikes, t = [], span.onset
            while t < span.end - EPS:
                if kind == "sustain" and t > span.onset:
                    break
                if t == span.onset or grid.strength(t) >= 0.75:
                    strikes.append(t)
                t += grid.beat
            for j, t in enumerate(strikes):
                stop = strikes[j + 1] if j + 1 < len(strikes) else span.end
                notes += [Note(t, stop - t, p, voice.vel, voice.name) for p in voicing]
        else:
            order, t, i = ORDER[kind], span.onset, 0
            while t < span.end - EPS:
                p = voicing[order[i % len(order)] % len(voicing)]
                notes.append(Note(t, min(voice.step, span.end - t), p, voice.vel, voice.name))
                t += voice.step
                i += 1
    return Events(notes, (0.0, plan.total_beats))


def realize_loop(voice, plan, chords, grid, rng) -> Events:
    """A short cell of scale degrees replayed on a rhythm across the whole piece."""
    lo, hi = voice.register
    notes = []
    for i, (onset, dur) in enumerate(voice.rhythm(0.0, plan.total_beats, grid, rng)):
        key = plan.key_at(onset)
        p = fold(key.degree(voice.cell[i % len(voice.cell)]), lo, hi)
        notes.append(Note(onset, dur, p, voice.vel, voice.name))
    return Events(notes, (0.0, plan.total_beats))


def realize(voice: Voice, plan: Plan, chords: ChordTrack, grid: MeterGrid, rng: Rng) -> Events:
    if voice.role == "melody":
        if voice.motif:
            from .motif import realize_motif_melody

            return realize_motif_melody(voice, plan, chords, grid, rng)
        return per_section(voice, plan, chords, grid, rng, realize_melody)
    if voice.role == "bass":
        return realize_bass(voice, plan, chords, grid, rng)
    if voice.role in ("accomp", "pad"):
        return realize_chords(voice, plan, chords, grid, rng)
    return realize_loop(voice, plan, chords, grid, rng)
```

```python
"""Motifs: a short idea developed into a period by transformation."""

from __future__ import annotations

from dataclasses import replace

from .events import Events, Note
from .harmony import ChordTrack
from .plan import Plan, Section
from .rhythm import MeterGrid
from .rng import Rng
from .theory import Scale
from .voices import EPS, Voice, realize_melody, requantize


def _invert_diatonic(motif: Events, key: Scale) -> Events:
    """Mirror scale degrees around the motif's first note."""
    axis = key.degree_of(motif[0].pitch)
    return motif.map(
        lambda n: replace(n, pitch=key.degree(2 * axis - key.degree_of(n.pitch)))
    )


def _cadences(ev: Events, voice: Voice, plan: Plan, sec: Section) -> Events:
    """Hold one long note on each cadence bar (end of phrase)."""
    bpb = plan.form.beats_per_bar
    n = sec.bars
    bars = [n - 1] + ([n // 2 - 1] if n >= 4 else [])
    notes = list(ev)
    for b in bars:
        t0 = sec.start + b * bpb
        t1 = t0 + bpb
        before = [x for x in notes if x.onset < t0 - EPS]
        if before:
            ref = before[-1].pitch
        elif notes:
            ref = notes[0].pitch
        else:
            ref = sum(voice.register) // 2
        notes = [x for x in notes if not (t0 - EPS <= x.onset < t1 - EPS)]
        notes = [replace(x, dur=t0 - x.onset) if x.end > t0 > x.onset else x for x in notes]
        notes.append(Note(t0, t1 - t0, ref, voice.vel, voice.name))
    return Events(notes, ev.span)


def develop_period(
    voice: Voice, plan: Plan, chords: ChordTrack, grid: MeterGrid, rng: Rng, sec: Section
) -> Events:
    """Motif -> sequence -> half cadence | motif -> variation -> authentic cadence."""
    bpb = plan.form.beats_per_bar
    key = plan.section_key(sec)
    mlen = (2 if sec.bars >= 8 else 1) * bpb
    half = (sec.bars // 2) * bpb if sec.bars >= 2 else sec.length
    motif = realize_melody(voice, plan, chords, grid, rng.child("motif"), sec, sec.start, sec.start + mlen)
    if not len(motif):
        lo, hi = voice.register
        motif = Events(
            [Note(sec.start, mlen, key.nearest((lo + hi) / 2), voice.vel, voice.name)],
            (sec.start, sec.start + mlen),
        )
    step = int(rng.choice([1, -1, 2]))
    variation = (
        _invert_diatonic(motif, key) if rng.choice(["invert", "augment"]) == "invert"
        else motif.augment(2)
    )
    parts: list[Events] = []
    t, k = 0.0, 0
    while t < half - EPS:
        parts.append((motif.transpose_diatonic(k * step, key) if k else motif).shift(t))
        t += mlen
        k += 1
    parts.append(motif.shift(half))
    t = half + mlen
    while t < sec.length - EPS:
        parts.append(variation.shift(t))
        t += variation.length
    notes = [
        replace(n, dur=min(n.dur, sec.end - n.onset))
        for p in parts
        for n in p
        if n.onset < sec.end - EPS
    ]
    return _cadences(Events(notes, (sec.start, sec.end)), voice, plan, sec)


def vary(ev: Events, sec: Section, plan: Plan) -> Events:
    """Primed sections (A'): lift the second half up a diatonic third."""
    key = plan.section_key(sec)
    half = sec.start + (sec.bars // 2) * plan.form.beats_per_bar
    return ev.map(
        lambda n: replace(n, pitch=key.degree(key.degree_of(n.pitch) + 2))
        if n.onset >= half - EPS
        else n
    )


def realize_motif_melody(
    voice: Voice, plan: Plan, chords: ChordTrack, grid: MeterGrid, rng: Rng
) -> Events:
    cache: dict[str, Events] = {}
    notes: list[Note] = []
    for sec in plan.form.sections:
        if sec.base_label not in cache:
            period = develop_period(voice, plan, chords, grid, rng.child(f"motif:{sec.base_label}"), sec)
            cache[sec.base_label] = period.shift(-sec.start)
        material = cache[sec.base_label].shift(sec.start)
        if sec.label != sec.base_label:
            material = vary(material, sec, plan)
        notes.extend(material)
    ev = Events(notes, (0.0, plan.total_beats))
```

```python
"""Performance: the human touch as curves (phrase arcs, ritardando, jitter)."""

from __future__ import annotations

from dataclasses import dataclass, replace

import numpy as np

from .curve import Curve, pink
from .events import Events
from .plan import Plan
from .rhythm import MeterGrid
from .rng import Rng


@dataclass
class Perform:
    """Todd's phrase-arc model plus a final ritardando and pink-noise timing."""

    jitter_ms: float = 8.0
    phrase_bars: int = 4
    tempo_arc: float = 0.06
    ritard: float = 0.25
    ritard_bars: int = 2
    dyn_depth: float = 0.3

    def arc(self, plan: Plan) -> Curve:
        """0 at phrase edges, 1 mid-phrase: 1 - 4(u - 1/2)^2."""
        L = self.phrase_bars * plan.form.beats_per_bar
        return Curve(lambda t: 1.0 - 4.0 * (np.mod(t, L) / L - 0.5) ** 2, "phrase_arc")

    def tempo(self, plan: Plan) -> Curve:
        """Push forward mid-phrase, and slow down over the final bars."""
        T = plan.total_beats
        R = min(self.ritard_bars * plan.form.beats_per_bar, T)
        rit = Curve(
            lambda t: 1.0 - self.ritard * np.clip((t - (T - R)) / R, 0.0, 1.0) ** 2,
            "ritard",
        )
        return plan.tempo * (1.0 + self.tempo_arc * (self.arc(plan) - 0.5)) * rit

    def apply(
        self, score: Events, plan: Plan, grid: MeterGrid, rng: Rng
    ) -> tuple[Events, Curve]:
        tempo = self.tempo(plan)
        arc = self.arc(plan)
        jitter: dict[str, Curve] = {}
        out = []
        for n in score:
            if n.voice not in jitter:
                jitter[n.voice] = pink(rng.child(n.voice), base=1 / 4)
            accent = 0.85 + 0.15 * grid.strength(n.onset)
            dyn = (1 - self.dyn_depth) + self.dyn_depth * float(arc(n.onset))
            tens = 0.7 + 0.3 * float(plan.tension(n.onset))
            vel = float(np.clip(n.vel * accent * dyn * tens, 0.05, 1.0))
            seconds = self.jitter_ms / 1000.0 * float(jitter[n.voice](n.onset))
            onset = max(score.span[0], n.onset + seconds * float(tempo(n.onset)) / 60.0)
            out.append(replace(n, onset=onset, vel=vel))
        return Events(out, score.span), tempo
```

```python
"""Additive synthesis: every timbre is a sum of decaying sine partials."""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np

from ..curve import Curve
from ..events import Events
from ..theory import freq

SR = 44100


@dataclass(frozen=True)
class Instrument:
    """Partials are (frequency ratio, amplitude, decay per second).

    ``inharmonicity`` is the string-stiffness constant B: partial n sounds at
    n*f0*sqrt(1 + B*n^2), slightly sharp, as on a real piano.
    """

    partials: tuple[tuple[float, float, float], ...]
    attack: float = 0.005
    release: float = 0.1
    inharmonicity: float = 0.0
    hammer: float = 0.0
    detune_cents: float = 0.0
    gain: float = 1.0


SINE = Instrument(((1.0, 1.0, 0.0),), attack=0.01, release=0.05)
EPIANO = Instrument(
    ((1, 1.0, 1.2), (2, 0.5, 1.8), (3, 0.25, 2.6), (4, 0.14, 3.4), (5, 0.08, 4.2), (6, 0.05, 5.0)),
    attack=0.003,
    release=0.15,
    inharmonicity=0.0004,
    hammer=0.04,
)
PAD = Instrument(
    ((1, 1.0, 0.05), (2, 0.45, 0.08), (3, 0.22, 0.12), (4, 0.12, 0.18)),
    attack=1.2,
    release=2.0,
    detune_cents=6.0,
    gain=0.6,
)
BELL = Instrument(
    ((1, 1.0, 0.7), (2.76, 0.6, 1.1), (5.4, 0.4, 1.7), (8.93, 0.25, 2.4)),
    attack=0.001,
    release=0.6,
    gain=0.7,
)


def render_note(inst: Instrument, pitch: int, vel: float, dur_s: float, sr: int = SR) -> np.ndarray:
    n = max(1, int(round((dur_s + inst.release) * sr)))
    t = np.arange(n) / sr
    f0 = freq(pitch)
    y = np.zeros(n)
    detunes = (0.0,) if inst.detune_cents == 0 else (-inst.detune_cents, inst.detune_cents)
    for ratio, amp, decay in inst.partials:
        f = f0 * ratio * np.sqrt(1.0 + inst.inharmonicity * ratio**2)
        for cents in detunes:
            fc = f * 2.0 ** (cents / 1200)
            if fc < sr / 2:
                y += (amp / len(detunes)) * np.sin(2 * np.pi * fc * t) * np.exp(-decay * t)
    if inst.hammer > 0:
        burst = np.random.default_rng(int(pitch)).standard_normal(n) * np.exp(-t / 0.004)
        y += inst.hammer * np.convolve(burst, np.ones(8) / 8, mode="same")
    env = np.minimum(1.0, t / inst.attack) if inst.attack > 0 else np.ones(n)
    if inst.release > 0:
        env = env * np.clip(1.0 - (t - dur_s) / inst.release, 0.0, 1.0)
    else:
        env = env * (t <= dur_s)
    return (y * env * inst.gain * vel**1.5).astype(np.float32)


def beats_to_seconds(tempo: Curve, beats) -> np.ndarray:
    """Seconds at each beat position: the integral of 60 / tempo(t) dt."""
    b = np.asarray(beats, dtype=float)
    top = float(np.max(b)) if b.size else 0.0
    grid = np.linspace(0.0, max(top, 1e-9), max(2, int(np.ceil(top * 64)) + 1))
    bpm = tempo(grid)
    if np.any(bpm <= 0):
        raise ValueError("tempo must stay positive")
    spb = 60.0 / bpm
    secs = np.concatenate([[0.0], np.cumsum((spb[1:] + spb[:-1]) / 2 * np.diff(grid))])
    return np.interp(b, grid, secs)


@dataclass(frozen=True)
class Reverb:
    """Convolution with exponentially decaying noise (-60 dB after ``seconds``)."""

    seconds: float = 1.5
    mix: float = 0.25
    seed: int = 0

    def impulse(self, sr: int = SR) -> np.ndarray:
        n = max(1, int(self.seconds * sr))
        t = np.arange(n) / sr
        ir = np.random.default_rng(self.seed).standard_normal(n) * np.exp(-6.91 * t / self.seconds)
        return ir / np.sqrt(np.sum(ir**2))

    def apply(self, y: np.ndarray, sr: int = SR) -> np.ndarray:
        ir = self.impulse(sr)
        size = len(y) + len(ir) - 1
        nfft = 1 << (size - 1).bit_length()
        wet = np.fft.irfft(np.fft.rfft(y, nfft) * np.fft.rfft(ir, nfft), nfft)[:size]
        dry = np.concatenate([y, np.zeros(len(ir) - 1)])
        return (1 - self.mix) * dry + self.mix * wet


def render(
    events: Events,
    tempo: Curve,
    instruments: Instrument | dict[str, Instrument] = EPIANO,
    reverb: Reverb | None = None,
    sr: int = SR,
) -> np.ndarray:
    def inst_for(voice: str) -> Instrument:
        if isinstance(instruments, dict):
            return instruments.get(voice, EPIANO)
        return instruments

    end_s = float(beats_to_seconds(tempo, [events.span[1]])[0])
    if len(events) == 0:
        return np.zeros(max(1, int(np.ceil(end_s * sr))), dtype=np.float32)
    ons = beats_to_seconds(tempo, [n.onset for n in events])
    offs = beats_to_seconds(tempo, [n.end for n in events])
    tail = max(inst_for(n.voice).release for n in events)
    total = int(np.ceil((max(end_s, float(offs.max())) + tail) * sr)) + 1
    out = np.zeros(total)
    for n, a, b in zip(events, ons, offs, strict=True):
        buf = render_note(inst_for(n.voice), n.pitch, n.vel, b - a, sr)
        i = int(round(a * sr))
        j = min(total, i + len(buf))
        out[i:j] += buf[: j - i]
    if reverb is not None:
        out = reverb.apply(out, sr)
    if not np.all(np.isfinite(out)):
        raise ValueError("synth produced non-finite audio")
    peak = float(np.max(np.abs(out)))
    if peak > 0:
        out = 0.98 * np.tanh(1.5 * out / peak) / np.tanh(1.5)
    return out.astype(np.float32)
```

```python
"""MIDI export: one track per voice plus a tempo map."""

from __future__ import annotations

import math
from pathlib import Path

import mido

from ..curve import Curve
from ..events import Events

_CHANNELS = [c for c in range(16) if c != 9]  # channel 9 is drums in General MIDI


def to_midi(events: Events, tempo: Curve, ppq: int = 480) -> mido.MidiFile:
    mf = mido.MidiFile(type=1, ticks_per_beat=ppq)
    conductor = mido.MidiTrack()
    mf.tracks.append(conductor)
    for b in range(int(math.ceil(events.span[1])) + 1):
        conductor.append(
            mido.MetaMessage(
                "set_tempo", tempo=mido.bpm2tempo(float(tempo(b))), time=0 if b == 0 else ppq
            )
        )
    for i, (voice, ev) in enumerate(sorted(events.by_voice().items())):
        track = mido.MidiTrack()
        mf.tracks.append(track)
        track.append(mido.MetaMessage("track_name", name=voice, time=0))
        ch = _CHANNELS[i % len(_CHANNELS)]
        msgs = []
        for n in ev:
            on = int(round(n.onset * ppq))
            off = max(on + 1, int(round(n.end * ppq)))
            pitch = min(127, max(0, n.pitch))
            vel = min(127, max(1, round(1 + 126 * n.vel)))
            msgs.append((on, 1, mido.Message("note_on", note=pitch, velocity=vel, channel=ch)))
            msgs.append((off, 0, mido.Message("note_off", note=pitch, velocity=0, channel=ch)))
        msgs.sort(key=lambda m: (m[0], m[1]))
        now = 0
        for tick, _, msg in msgs:
            track.append(msg.copy(time=tick - now))
            now = tick
    return mf


def save_midi(events: Events, tempo: Curve, path) -> Path:
    path = Path(path)
    to_midi(events, tempo).save(str(path))
    return path
```

```python
"""Piece: runs Plan -> Harmony -> Voices -> Perform -> Render, caching each layer."""

from __future__ import annotations

from functools import cached_property
from pathlib import Path

import numpy as np

from .curve import Curve
from .events import Events
from .harmony import ChordTrack
from .perform import Perform
from .plan import Plan
from .render.midi import save_midi
from .render.play import write_wav
from .render.synth import EPIANO, Instrument, Reverb, beats_to_seconds, render
from .rhythm import MeterGrid
from .voices import Voice, realize

_DEFAULT_REVERB = Reverb()


class Piece:
    def __init__(
        self,
        plan: Plan,
        harmony,
        voices: list[Voice],
        perform: Perform | None = None,
        instruments: Instrument | dict[str, Instrument] = EPIANO,
        reverb: Reverb | None = _DEFAULT_REVERB,
    ):
        names = [v.name for v in voices]
        if not names:
            raise ValueError("a piece needs at least one voice")
        if len(set(names)) != len(names):
            raise ValueError(f"voice names must be unique, got {names}")
        self.plan = plan
        self.harmony = harmony
        self.voices = list(voices)
        self.perform = perform if perform is not None else Perform()
        self.instruments = instruments
        self.reverb = reverb
        self.grid = MeterGrid(plan.form.meter)

    @cached_property
    def chords(self) -> ChordTrack:
        return self.harmony.generate(self.plan, self.plan.rng.child("harmony"))

    @cached_property
    def score(self) -> Events:
        out = Events(span=(0.0, self.plan.total_beats))
        for v in self.voices:
            rng = self.plan.rng.child(f"voice:{v.name}")
            out = out | realize(v, self.plan, self.chords, self.grid, rng)
        return out

    @cached_property
    def _performed(self) -> tuple[Events, Curve]:
        return self.perform.apply(
            self.score, self.plan, self.grid, self.plan.rng.child("perform")
        )

    @property
    def events(self) -> Events:
        return self._performed[0]

    @property
    def tempo(self) -> Curve:
        return self._performed[1]

    @property
    def duration(self) -> float:
        return float(beats_to_seconds(self.tempo, [self.plan.total_beats])[0])

    def voice(self, name: str) -> Events:
        return self.score.filter(lambda n: n.voice == name)

    @cached_property
    def _audio(self) -> np.ndarray:
        return render(self.events, self.tempo, self.instruments, self.reverb)

    def audio(self) -> np.ndarray:
        return self._audio

    def play(self) -> None:
        from .render import play as _play

        _play.play(self.audio())

    def show(self, save=None):
        from .render.viz import piano_roll

        return piano_roll(self, save=save)

    def save_wav(self, path) -> Path:
        return write_wav(path, self.audio())

    def save_midi(self, path) -> Path:
        return save_midi(self.events, self.tempo, path)

    def __repr__(self) -> str:
        return (
            f"Piece({self.plan.form.spec!r}, key={self.plan.key.name} "
            f"{self.plan.key.root}, voices={[v.name for v in self.voices]})"
        )
```

```python
def plot_curve(curve, t0: float = 0.0, t1: float = 1.0, n: int = 512) -> Figure:
    fig = Figure(figsize=(7, 2.5), constrained_layout=True)
    ax = fig.add_subplot()
    t, y = curve.sample(t0, t1, n)
    ax.plot(t, y, color=PALETTE[0])
    ax.set_title(curve.name, fontsize=9)
    ax.set_xlabel("t (beats)")
    ax.grid(alpha=0.3)
    return fig


def plot_chords(track) -> Figure:
    fig = Figure(figsize=(10, 1.6), constrained_layout=True)
    ax = fig.add_subplot()
    for s in track:
        ax.barh(0, s.dur, left=s.onset, color=FUNCTION_COLORS.get(s.function, "#888888"),
                alpha=0.3, edgecolor="white")
        ax.text(s.onset + s.dur / 2, 0, str(s.chord), ha="center", va="center", fontsize=7)
    ax.set_yticks([])
    ax.set_xlabel("beats")
    return fig


def piano_roll(piece, save=None) -> Figure:
    events, plan = piece.events, piece.plan
    T = plan.total_beats
    pitches = [n.pitch for n in events] or [60]
    lo, hi = min(pitches) - 2, max(pitches) + 3

    fig = Figure(figsize=(14, 7), constrained_layout=True)
    gs = fig.add_gridspec(2, 2, width_ratios=[0.6, 14], height_ratios=[6, 1])
    ax = fig.add_subplot(gs[0, 1])
    kb = fig.add_subplot(gs[0, 0], sharey=ax)
    ten = fig.add_subplot(gs[1, 1], sharex=ax)
    fig.add_subplot(gs[1, 0]).axis("off")

    for s in plan.form.sections:
        if s.index % 2:
            ax.axvspan(s.start, s.end, color="#000000", alpha=0.04, lw=0)
        ax.text(s.start + 0.3, hi + 1.2, s.label, fontsize=10, weight="bold", va="bottom")

    for i, v in enumerate(piece.voices):
        notes = [n for n in events if n.voice == v.name]
        if notes:
            verts = [
                [(n.onset, n.pitch - 0.4), (n.end, n.pitch - 0.4),
                 (n.end, n.pitch + 0.4), (n.onset, n.pitch + 0.4)]
                for n in notes
            ]
            colors = [to_rgba(_color(i), 0.25 + 0.75 * n.vel) for n in notes]
            ax.add_collection(PolyCollection(verts, facecolors=colors, edgecolors="none", label=v.name))

    t = np.linspace(0, T, 800)
    for i, v in enumerate(piece.voices):
        if v.contour is not None:
            a, b = v.register
            ax.plot(t, a + v.contour(t) * (b - a), color=_color(i), lw=1, ls="--", alpha=0.8,
                    label=f"{v.name} contour")

    for s in piece.chords:
        pcs = ",".join(str(p) for p in sorted(s.chord.pcs))
        ax.text(s.onset + 0.1, hi + 0.9, f"{s.chord}\n{{{pcs}}}", fontsize=6, va="top", color="#444444")

    ax.set_xlim(0, T)
    ax.set_ylim(lo, hi + 3)
    ax.tick_params(labelleft=False)
    ax.legend(loc="lower right", fontsize=7)

    scale_pcs = plan.key.pcs
    for p in range(lo, hi + 3):

    kb.set_xlim(0, 1)
    kb.set_xticks([])
    kb.set_ylabel("MIDI pitch")

    tt = plan.tension(t)
    ten.fill_between(t, 0, tt, color=PALETTE[1], alpha=0.25)
    ten.plot(t, tt, color=PALETTE[1], lw=1)
    ten.set_ylim(0, 1.05)
    ten.set_ylabel("T(t)")
    ten.set_xlabel("beats")

    if save is not None:
        fig.savefig(save, bbox_inches="tight", dpi=120)
    return fig
```

```python
from __future__ import annotations

from .curve import arch, pink, sine
from .harmony import Functional, TonnetzWalk
from .perform import Perform
from .piece import Piece
from .plan import Form, Plan
from .render.synth import BELL, EPIANO, PAD, Reverb
from .rhythm import density, euclid
from .rng import Rng
from .rules import avoid_repeat, end_on_tonic, gap_fill, resolve_leading_tone, step_bias
from .theory import Scale
from .voices import Voice

OVERRIDES = (
    "key", "meter", "tempo", "form", "plan", "harmony", "voices", "perform",
    "instruments", "reverb",
)
AMBIENT_MODES = ("dorian", "lydian", "aeolian", "pentatonic")


def _check(overrides: dict) -> None:
    unknown = set(overrides) - set(OVERRIDES)
    if unknown:
        raise TypeError(f"unknown override(s) {sorted(unknown)}; use {OVERRIDES}")


def tonal(seed: int = 0, **overrides) -> Piece:
    """A classical-style piano piece: A A' B A, functional harmony, a developed motif."""
    _check(overrides)
    rng = Rng(seed).child("preset:tonal")
    root = 60 + int(rng.integers(-5, 7))
    minor = rng.random() < 0.35
    triple = rng.random() < 0.3
    tempo_choice = float(rng.uniform(72, 100))

    key = overrides.get("key", Scale.minor(root) if minor else Scale.major(root))
    meter = overrides.get("meter", (3, 4) if triple else (4, 4))
    tempo = overrides.get("tempo", tempo_choice)
    form = overrides.get("form", Form("A A' B A", bars=8, meter=meter))
    plan = overrides.get("plan", Plan(seed, key=key, form=form, tempo=tempo, modulate=True))
    bpb = plan.form.beats_per_bar

    # Melody shape: an arch per 4-bar phrase, plus 1/f wandering.
    contour = (
        0.55 * arch().tile(4 * bpb) + 0.2 + 0.25 * pink(rng.child("contour"), base=1 / 8)
    ).clip(0.0, 1.0)
    voices = overrides.get(
        "voices",
        [
            Voice(
                "melody", "melody", (60, 84),
                rhythm=density(0.85, step=0.5),
                contour=contour,
                rules=[
                    step_bias(1.5), gap_fill(4.0), avoid_repeat(2.0, 2),
                    resolve_leading_tone(3.0), end_on_tonic(20.0),
                ],
                motif=True,
                vel=0.8,
            ),
            Voice("bass", "bass", (36, 60), vel=0.6),
            Voice("accomp", "accomp", (48, 67), pattern={"A": "alberti", "B": "block"}, vel=0.42),
        ],
    )
    return Piece(
        plan,
        overrides.get("harmony", Functional()),
        voices,
        overrides.get("perform", Perform(jitter_ms=6.0)),
        overrides.get("instruments", EPIANO),
        overrides.get("reverb", Reverb(seconds=1.2, mix=0.18)),
    )


def ambient(seed: int = 0, **overrides) -> Piece:
    """A slow modal piece: Tonnetz harmony, a sparse melody, two phasing loops."""
    _check(overrides)
    rng = Rng(seed).child("preset:ambient")
    mode = rng.choice(AMBIENT_MODES)
    root = 57 + int(rng.integers(0, 8))
    tempo_choice = float(rng.uniform(50, 66))

    key = overrides.get("key", Scale.mode(mode, root))
    meter = overrides.get("meter", (4, 4))
    tempo = overrides.get("tempo", tempo_choice)
    form = overrides.get("form", Form("A B A' C A", bars=8, meter=meter))
    plan = overrides.get("plan", Plan(seed, key=key, form=form, tempo=tempo))

    subset = (0, 1, 2, 4, 5) if len(plan.key) == 7 else None  # pentatonic subset of a mode
    contour = (
        0.5 + 0.25 * sine(1 / 32) + 0.25 * pink(rng.child("contour"), base=1 / 16)
    ).clip(0.0, 1.0)
    voices = overrides.get(
        "voices",
        [
            Voice("pad", "pad", (48, 72), vel=0.35),
            Voice(
                "melody", "melody", (62, 86),
                rhythm=density(0.35, step=1.0, max_dur=4.0),
                contour=contour,
                rules=[step_bias(1.0), gap_fill(3.0), avoid_repeat(2.0, 3)],
                temperature=0.8,
                subset=subset,
                vel=0.5,
            ),
            # Loops of 8 and 13 steps line up again only every lcm(8, 13) = 104 steps.
            Voice("loop1", "loop", (67, 91), rhythm=euclid(3, 8, step=0.5, dur=1.0),
                  cell=(0, 2, 4), vel=0.28),
            Voice("loop2", "loop", (55, 79), rhythm=euclid(5, 13, step=0.5, dur=1.0),
                  cell=(4, 3, 1, 0, 2), vel=0.22),
        ],
    )
    return Piece(
        plan,
        overrides.get("harmony", TonnetzWalk(bars_per_chord=(2, 4))),
        voices,
        overrides.get("perform", Perform(jitter_ms=15.0, tempo_arc=0.03, dyn_depth=0.2, ritard=0.3)),
        overrides.get("instruments", {"pad": PAD, "melody": EPIANO, "loop1": BELL, "loop2": BELL}),
        overrides.get("reverb", Reverb(seconds=3.5, mix=0.4)),
    )
```

````markdown
# sinfony

Music from mathematics. A piece is written as layered math (curves, noise,
and real music theory), and sinfony turns it into sound, MIDI, and an
annotated piano roll.

```python
import sinfony as sf

piece = sf.tonal(seed=42)   # or sf.ambient(seed=42)
piece.play()                # inline in Jupyter, speakers, or a WAV file
piece.show()                # piano roll with contour, chords, and tension
piece.save_midi("tonal.mid")
```

## How it works

| Layer | Math |
|---|---|
| Plan | form (A A' B A), key areas, tension curve T(t) |
| Harmony | functional Markov chain (T→S→D→T) or a Tonnetz walk (P, L, R) |
| Voices | contour curve + 1/f noise → nearest allowed pitch, shaped by rule penalties |
| Motifs | transpose, invert, augment: group actions on notes |
| Perform | phrase arcs (Todd), ritardando, pink-noise micro-timing |
| Synth | additive partials; piano inharmonicity fₙ = n·f₀·√(1+B·n²) |

Same seed → same piece. Every layer can be inspected (`piece.chords`,
`piece.score`, `piece.plan.tension`).
````
